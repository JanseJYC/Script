local event = {} do
    function event.new()
        local self = setmetatable({Alive = true}, {__call = function(s, ...) s:Fire(...) end})
        local BindableEvent = Instance.new("BindableEvent")
        function self:Connect(callback)
            local c = {}
            local conn = BindableEvent.Event:Connect(callback)
            c.Connected = true
            function c:Disconnect()
                conn:Disconnect()
                c.Connected = false
            end
            return c
        end
        function self:Fire(...) BindableEvent:Fire(...) end
        function self:Destroy()
            self.Alive = false
            BindableEvent:Destroy()
        end
        return self
    end
end

local mouse = {} do
    mouse.held = false
    mouse.InputBegan = event.new()
    mouse.InputEnded = event.new()
    UserInputService.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            mouse.held = true
            mouse.InputBegan:Fire()
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            mouse.held = false
            mouse.InputEnded:Fire()
        end
    end)
end

end

local function resize(part, props, delay)

end

local windowHistory = {}
local windowCache = {}
local mouseCache = {}
local browsingWindow = {}

local function updateWindowHistory()
    for win, lvl in next, windowHistory do
        local offset = 9000 - lvl * 100
        win.ZIndex = (windowCache[win][win] or 0) + offset
        for _, desc in next, win:GetDescendants() do
            if desc:IsA("GuiObject") and windowCache[win][desc] then
                desc.ZIndex = windowCache[win][desc] + offset
            end
        end
    end
end

local function cacheWindowHistory(win)
    windowCache[win] = {}
    windowCache[win][win] = win.ZIndex
    for _, desc in next, win:GetDescendants() do
        if desc:IsA("GuiObject") then
            windowCache[win][desc] = desc.ZIndex
        end
    end
    win.DescendantAdded:Connect(function(d)
        if d:IsA("GuiObject") then
            windowCache[win][d] = d.ZIndex
            updateWindowHistory()
        end
    end)
end

local function setTopMost(win)
    for w, l in next, windowHistory do
        if w ~= win then windowHistory[w] = l + 1 end
    end
    windowHistory[win] = 1
    updateWindowHistory()
end

local function findBrowsingTopMost()
    local top, lvl = nil, math.huge
    for w, l in next, windowHistory do
        if (mouseCache[w] or browsingWindow[w]) and l < lvl then
            lvl = l
            top = w
        end
    end
    return top
end

local dragger = {} do
    local draggerCache = {}
    function dragger.new(frame)
        frame.Active = true
        frame.MouseEnter:Connect(function() draggerCache[frame] = true end)
        frame.MouseLeave:Connect(function() draggerCache[frame] = false end)
        mouse.InputBegan:Connect(function()
            if findBrowsingTopMost() == frame and not colorpicking and not sliding and draggerCache[frame] then
                local offset = getMouse() - Vector2.new(frame.AbsolutePosition.X, frame.AbsolutePosition.Y)
                while mouse.held do
                    local pos = getMouse() - offset
                    frame.Position = UDim2.new(0, pos.X, 0, pos.Y)
                    RunService.Heartbeat:Wait()
                end
            end
        end)
    end
end

local settings = {new = function(def)
    return {handle = function(opts)
        opts = typeof(opts) == "table" and opts or {}
        local res = {}
        for k, v in next, def do res[k:lower()] = v end
        for k, v in next, opts do
            if typeof(def[k:lower()]) == typeof(v) then res[k:lower()] = v end
        end
        return res
    end}
end}

            function types.slider(sOpts)
                local s = {}

                sl.Size = UDim2.new(0, sOpts.size + 8 + txt.TextBounds.X, 0, 20)
                local function setPos(p) resize(bar, {Position = UDim2.new(p, -2.5, 0, 0)}, sOpts.animation) end
                val.Text = round(s.value, 2)
                local oldVal
                function s.set(n)
                    assert(typeof(n) == "number")
                    local mn, mx = sOpts.min, sOpts.max
                    if mx < mn then mn, mx = mx, mn end
                    n = math.clamp(n, mn, mx)
                    s.value = n
                    if s.value ~= oldVal and not s.eventBlock then s.event:Fire(s.value) end
                    oldVal = s.value
                    val.Text = round(s.value, 2)
                    local p = (n - mn) / (mx - mn)
                    setPos(p)
                end
                s.set(s.value)
                local inside = false
                inner.MouseEnter:Connect(function() inside = true end)
                inner.MouseLeave:Connect(function() inside = false end)
                mouse.InputBegan:Connect(function()
                    if inside and findBrowsingTopMost() == main then
                        while mouse.held do
                            sliding = true
                            local p = getMouse()
                            local x = math.clamp(p.X - inner.AbsolutePosition.X, 0, sOpts.size)
                            local m = sOpts.size / math.abs(sOpts.max - sOpts.min)
                            local v = math.floor(x / m)
                            s.set(v + (sOpts.min < sOpts.max and sOpts.min or sOpts.max))
                            RunService.Heartbeat:Wait()
                        end
                    end
                end)
                mouse.InputEnded:Connect(function() sliding = false end)
                function s:Destroy() sl:Destroy() end
                updateCanvas()
                s.options = sOpts
                s.self = sl
                return s
            end

local function rgbtohsv(color)
    local r, g, b = color.r, color.g, color.b
    local max, min = math.max(r,g,b), math.min(r,g,b)
    local h, s, v = 0, 0, max
    local d = max - min
    if max ~= 0 then s = d / max end
    if max == min then h = 0
    elseif max == r then h = (g - b) / d + (g < b and 6 or 0)
    elseif max == g then h = (b - r) / d + 2
    elseif max == b then h = (r - g) / d + 4 end
    h = h / 6
    return h, s, v
end
